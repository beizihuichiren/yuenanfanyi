## 一、校验层（Validation Layer）

校验层是入口关卡，所有数据进处理层之前必须通过。校验失败直接打回，不进入后续环节。

### 1.1 视频元信息提取

```python
import subprocess, json, re, hashlib
from dataclasses import dataclass, field
from pathlib import Path

@dataclass
class VideoMeta:
    path: Path
    codec: str          = ""
    width: int          = 0
    height: int         = 0
    duration: float     = 0.0     # 秒
    fps: float          = 0.0
    bitrate: int        = 0       # kbps
    audio_codec: str    = ""
    audio_channels: int = 0
    audio_sample_rate: int = 0
    file_size: int      = 0
    has_audio: bool     = False
    errors: list[str]   = field(default_factory=list)


def probe_video(path: Path) -> VideoMeta | None:
    """
    用 ffprobe 提取视频元信息。
    解析失败 → 文件损坏或不是有效视频 → 返回 None。
    """
    try:
        r = subprocess.run([
            "ffprobe", "-v", "quiet", "-print_format", "json",
            "-show_format", "-show_streams", str(path)
        ], capture_output=True, text=True, timeout=30)
        data = json.loads(r.stdout)
    except (subprocess.TimeoutExpired, json.JSONDecodeError):
        return None

    video = audio = None
    for s in data.get("streams", []):
        if s["codec_type"] == "video" and video is None:
            video = s
        elif s["codec_type"] == "audio" and audio is None:
            audio = s
    if video is None:
        return None

    fmt = data.get("format", {})
    fps_str = video.get("r_frame_rate", "0/1")
    num, den = map(int, fps_str.split("/"))
    return VideoMeta(
        path       = path,
        codec      = video.get("codec_name", ""),
        width      = video.get("width", 0),
        height     = video.get("height", 0),
        duration   = float(fmt.get("duration", 0)),
        fps        = num / den if den else 0,
        bitrate    = int(fmt.get("bit_rate", 0)) // 1000,
        audio_codec= audio.get("codec_name", "") if audio else "",
        audio_channels = audio.get("channels", 0) if audio else 0,
        audio_sample_rate = audio.get("sample_rate", 0) if audio else 0,
        file_size  = path.stat().st_size,
        has_audio  = audio is not None,
    )
```

### 1.2 文件命名与完整性校验

```python
NAMING_PATTERN = re.compile(
    r"^(?P<name>.+?)_EP(?P<ep>\d{2})_(?P<usage>.+?)\.(?P<ext>mp4|mkv|mov)$"
)

def validate_naming(files: list[Path], expected_name: str, ep_range: range) -> list[str]:
    """
    规则：
    - 文件名必须匹配 {剧名}_EP{两位数集数}_{用途}.{扩展名}
    - 剧名与 expected_name 模糊匹配（忽略空格和下划线差异）
    - 集数必须在 ep_range 内且连续无缺失
    - 同一集不允许重复文件
    """
    errors = []
    found = set()
    name_key = expected_name.lower().replace(" ", "_")

    for f in files:
        m = NAMING_PATTERN.match(f.name)
        if not m:
            errors.append(f"命名不规范: {f.name}")
            continue
        file_name_key = m.group("name").lower().replace(" ", "_")
        if file_name_key != name_key:
            errors.append(f"剧名不匹配: {f.name} (期望 {expected_name})")
        ep = int(m.group("ep"))
        if ep not in ep_range:
            errors.append(f"集数超出范围: {f.name}")
        if ep in found:
            errors.append(f"重复集数: {f.name}")
        found.add(ep)

    missing = set(ep_range) - found
    if missing:
        errors.append(f"缺集: {sorted(missing)}")
    return errors
```

### 1.3 版本一致性校验

```python
def validate_version_consistency(metas: list[VideoMeta]) -> list[str]:
    """
    同剧各集必须来自同一片源版本，否则字幕时间轴会整体错位：
    - 分辨率、帧率、编码必须一致
    - 时长偏离中位数超过 10% 告警（可能缺片段）
    - 片头黑屏时长偏差 > 2s 告警（可能版本不同）
    """
    errors = []
    base = metas[0]
    for m in metas[1:]:
        if (m.width, m.height) != (base.width, base.height):
            errors.append(f"{m.path.name}: 分辨率不一致 ({m.width}x{m.height} vs {base.width}x{base.height})")
        if abs(m.fps - base.fps) > 0.05:
            errors.append(f"{m.path.name}: 帧率不一致 ({m.fps:.2f} vs {base.fps:.2f})")
        if m.codec != base.codec:
            errors.append(f"{m.path.name}: 编码不一致 ({m.codec} vs {base.codec})")

    # 时长偏离检测
    durations = [m.duration for m in metas if m.duration > 0]
    if durations:
        median = sorted(durations)[len(durations) // 2]
        for m in metas:
            if m.duration > 0 and abs(m.duration - median) / median > 0.1:
                errors.append(f"{m.path.name}: 时长偏离 ({m.duration:.0f}s vs 中位数 {median:.0f}s)")

    # 片头黑屏检测（快速版本差异指标）
    head_blacks = []
    for m in metas:
        blacks = detect_black_frames(m.path)
        head = next((b for b in blacks if b[0] < 1), None)
        head_blacks.append(head[1] if head else 0)
    if head_blacks:
        median_head = sorted(head_blacks)[len(head_blacks) // 2]
        for i, h in enumerate(head_blacks):
            if abs(h - median_head) > 2:
                errors.append(f"{metas[i].path.name}: 片头长度不一致 ({h:.1f}s vs {median_head:.1f}s)")

    return errors
```

```python
def detect_black_frames(path: Path, dur=1.0, pix=0.10) -> list[tuple[float, float]]:
    """
    FFmpeg blackdetect。dur=1s 过滤正常转场黑场，pix=0.10 像素亮度 < 10% 算黑。
    返回 [(start, end), ...]，单位秒。
    """
    r = subprocess.run([
        "ffmpeg", "-i", str(path),
        "-vf", f"blackdetect=d={dur}:pix_th={pix}",
        "-an", "-f", "null", "-"
    ], capture_output=True, text=True)

    segs = []
    for line in r.stderr.split("\n"):
        if "black_start" in line:
            s = float(re.search(r"black_start:([\d.]+)", line).group(1))
            e = float(re.search(r"black_end:([\d.]+)", line).group(1))
            segs.append((s, e))
    return segs
```

```python
import numpy as np

def detect_glitch_frames(path: Path, sample_interval=0.5, psnr_thresh=15.0):
    """
    每 0.5s 抽一帧，与前一帧算 PSNR。
    正常帧间 PSNR 30-50dB。连续 2 帧 < 15dB 且不在场景切换密集区 → 疑似花屏。
    返回时间点列表。
    """
    r = subprocess.run([
        "ffmpeg", "-i", str(path),
        "-vf", f"select='not(mod(n,{int(sample_interval*25)}))',setpts=N/FRAME_RATE/TB",
        "-vsync", "vfr", "-f", "image2pipe", "-vcodec", "rawvideo",
        "-pix_fmt", "rgb24", "-"
    ], capture_output=True, timeout=120)

    w, h = 1920, 1080   # 用 ffprobe 获取实际值
    frame_size = w * h * 3
    frames = []
    raw = r.stdout
    for i in range(0, len(raw), frame_size):
        chunk = raw[i:i+frame_size]
        if len(chunk) == frame_size:
            frames.append(np.frombuffer(chunk, dtype=np.uint8).reshape((h, w, 3)))

    glitches = []
    for i in range(1, len(frames)):
        mse = np.mean((frames[i].astype(float) - frames[i-1].astype(float)) ** 2)
        psnr = 20 * np.log10(255.0 / np.sqrt(mse)) if mse > 0 else 100
        if psnr < psnr_thresh and (i < 2 or psnr_prev > psnr_thresh):
            glitches.append(i * sample_interval)
        psnr_prev = psnr

    # 过滤正常场景切换：密集跳变区（3s 内 > 3 次）视为正常切换
    return _filter_cluster(glitches, window=3, max_count=3)


def _filter_cluster(times: list[float], window=3, max_count=3):
    """窗口内超过 max_count 个点视为正常密集事件，过滤掉"""
    if len(times) <= max_count:
        return times
    result = []
    i = 0
    while i < len(times):
        j = i
        while j < len(times) and times[j] - times[i] <= window:
            j += 1
        if j - i <= max_count:
            result.extend(times[i:j])
        i = j
    return result
```

```python
def detect_black_bars(path: Path, sample_count=10) -> dict:
    """
    均匀抽 sample_count 帧，检测四边是否有持续黑边。
    黑边判定：边缘 5% 区域像素均值 < 15。
    """
    # 抽样关键帧
    r = subprocess.run([
        "ffmpeg", "-i", str(path),
        "-vf", f"select='not(mod(n,{max(1, int(600/sample_count))}))'",
        "-vsync", "vfr", "-frames:v", str(sample_count),
        "-f", "image2pipe", "-vcodec", "rawvideo", "-pix_fmt", "rgb24", "-"
    ], capture_output=True, timeout=30)

    # 用 ffprobe 获取实际分辨率（此处省略，假设 1920x1080）
    w, h = 1920, 1080
    edge_pct = 0.05
    top_h = int(h * edge_pct)
    bot_h = int(h * edge_pct)
    left_w = int(w * edge_pct)
    right_w = int(w * edge_pct)

    top_black = bot_black = left_black = right_black = 0

    raw = r.stdout
    frame_size = w * h * 3
    n_frames = len(raw) // frame_size
    for fi in range(n_frames):
        arr = np.frombuffer(raw[fi*frame_size:(fi+1)*frame_size], dtype=np.uint8).reshape((h, w, 3))
        if arr[:top_h, :, :].mean() < 15:   top_black   += 1
        if arr[-bot_h:, :, :].mean() < 15:   bot_black   += 1
        if arr[:, :left_w, :].mean() < 15:   left_black  += 1
        if arr[:, -right_w:, :].mean() < 15:  right_black += 1

    return {
        "top":    top_black / max(n_frames, 1),
        "bottom": bot_black / max(n_frames, 1),
        "left":   left_black / max(n_frames, 1),
        "right":  right_black / max(n_frames, 1),
    }
```

```python
def detect_av_sync(path: Path) -> bool:
    """
    最简单方案：分别提取音频和视频的静音/黑屏时间线，
    对比第一个非静音和非黑屏的时间点差异。> 0.5s 判定不同步。
    """
    # 提取视频黑屏区间
    blacks = detect_black_frames(path, dur=0.3, pix=0.08)

    # 提取音频静音区间
    r = subprocess.run([
        "ffmpeg", "-i", str(path),
        "-af", "silencedetect=n=-50dB:d=0.3",
        "-f", "null", "-"
    ], capture_output=True, text=True)

    silences = []
    silence_start = None
    for line in r.stderr.split("\n"):
        if "silence_start" in line:
            silence_start = float(re.search(r"silence_start:\s*([\d.]+)", line).group(1))
        elif "silence_end" in line and silence_start is not None:
            silence_end = float(re.search(r"silence_end:\s*([\d.]+)", line).group(1))
            silences.append((silence_start, silence_end))
            silence_start = None

    # 对比：取第一个非黑屏时刻 vs 第一个非静音时刻
    first_non_black = _find_first_non_black(blacks, 100)
    first_non_silent = _find_first_non_silent(silences, 100)
    return abs(first_non_black - first_non_silent) <= 0.5


def _find_first_non_black(segments, max_dur):
    """找到第一个不处于黑屏段的时间点"""
    if not segments:
        return 0
    pos = 0.0
    for s, e in segments:
        if s <= pos < e:
            pos = e
        elif pos < s:
            return pos
    return pos


def _find_first_non_silent(segments, max_dur):
    return _find_first_non_black(segments, max_dur)
```

## 二、清洗层（Cleaning Layer）

校验通过后的数据进入清洗层。清洗层处理的是数据本身的质量问题，不改动业务语义。

### 2.1 SRT 格式解析

```python
import re
from dataclasses import dataclass

@dataclass
class SrtEntry:
    index: int
    start: float      # 秒
    end: float        # 秒
    text: str

SRT_TIME = re.compile(r"(\d{2}):(\d{2}):(\d{2})[,.](\d{3})")

def parse_srt(path: Path) -> list[SrtEntry]:
    """解析 SRT 文件，返回结构化条目列表"""
    text = path.read_text(encoding="utf-8")
    entries = []
    blocks = text.strip().split("\n\n")

    for block in blocks:
        lines = block.strip().split("\n")
        if len(lines) < 3:
            continue
        try:
            idx = int(lines[0].strip())
        except ValueError:
            continue

        time_match = re.findall(SRT_TIME, lines[1])
        if len(time_match) != 2:
            continue
        start = _time_to_seconds(time_match[0])
        end   = _time_to_seconds(time_match[1])
        text  = "\n".join(lines[2:])
        entries.append(SrtEntry(index=idx, start=start, end=end, text=text))

    return entries


def _time_to_seconds(t: tuple) -> float:
    return int(t[0]) * 3600 + int(t[1]) * 60 + int(t[2]) + int(t[3]) / 1000.0
```

### 2.2 SRT 格式修复

```python
def repair_srt(entries: list[SrtEntry]) -> list[SrtEntry]:
    """
    清洗规则：
    1. 序号重新从 1 开始连续编号
    2. 过滤空文本条目
    3. 合并时间轴重叠的连续条目（如「我马上」+「来」→「我马上来」）
    4. 拆分过长的单条字幕（> 40 字符 → 在标点处拆分）
    5. 修正 end < start 的异常时间轴
    """
    # 规则1: 去空
    entries = [e for e in entries if e.text.strip()]

    # 规则2: 修正异常时间轴
    for e in entries:
        if e.end <= e.start:
            e.end = e.start + 1.0   # 兜底给 1 秒

    # 规则3: 合并时间轴重叠的相邻条目
    merged = []
    for e in entries:
        if merged and e.start < merged[-1].end:
            prev = merged[-1]
            if e.text.strip() == prev.text.strip():
                continue   # 完全重复，丢弃
            prev.text = prev.text.rstrip() + e.text.lstrip()
            prev.end = max(prev.end, e.end)
        else:
            merged.append(e)

    # 规则4: 拆分过长的条目
    result = []
    MAX_CHARS = 40
    for e in merged:
        if len(e.text) <= MAX_CHARS:
            result.append(e)
        else:
            parts = _split_long_text(e.text, MAX_CHARS)
            seg_dur = (e.end - e.start) / len(parts)
            for i, p in enumerate(parts):
                result.append(SrtEntry(
                    index=0,   # 后续重新编号
                    start=e.start + i * seg_dur,
                    end=e.start + (i + 1) * seg_dur,
                    text=p,
                ))
            # 补充：如果 e 在 merged 中是中间元素，它没有 index 也没关系，后面统一重编号

    # 规则5: 重新编号
    for i, e in enumerate(result):
        e.index = i + 1

    return result


def _split_long_text(text: str, max_chars: int) -> list[str]:
    """在标点处拆分，尽量均匀"""
    parts = []
    remaining = text
    while len(remaining) > max_chars:
        split_at = max_chars
        for punct in "，。！？；、\n":
            pos = remaining.rfind(punct, 0, max_chars)
            if pos > max_chars * 0.5:
                split_at = pos + 1
                break
        parts.append(remaining[:split_at].strip())
        remaining = remaining[split_at:].strip()
    if remaining:
        parts.append(remaining)
    return parts
```

### 2.3 字幕时间轴去噪

```python
def denoise_timeline(entries: list[SrtEntry], min_gap=0.05, min_dur=0.3):
    """
    时间轴微调：
    - 相邻条目间距 < min_gap → 拉近到无缝衔接
    - 条目时长 < min_dur → 扩展至 min_dur（向后借时间，不覆盖下一条）
    - 条目时长 > 8s → 标记告警（可能在清洗层标记但不强制拆分）
    """
    for i in range(len(entries)):
        # 扩展过短的字幕
        if entries[i].end - entries[i].start < min_dur:
            new_end = entries[i].start + min_dur
            if i + 1 < len(entries) and new_end > entries[i + 1].start - min_gap:
                new_end = entries[i + 1].start - min_gap
            entries[i].end = new_end

        # 消除过小间隔
        if i + 1 < len(entries):
            gap = entries[i + 1].start - entries[i].end
            if 0 < gap < min_gap:
                entries[i].end = entries[i + 1].start
```

### 2.5 双路交叉校验（剪映 + Whisper）

```python
from difflib import SequenceMatcher

def crosscheck(cn_srt: list[SrtEntry], whisper_srt: list[SrtEntry]) -> dict:
    """
    剪映 SRT 和 Whisper SRT 逐条对齐比对。
    时间轴对齐策略：按 start 时间做最近邻匹配。

    返回三类标记：
    - green:  文本相似度 >= 0.9  → 跳过
    - yellow: 0.7 <= 相似度 < 0.9 → 抽查
    - red:    相似度 < 0.7 或某条缺失 → 必须人工核对
    """
    report = {"green": [], "yellow": [], "red": []}

    # 时间轴对齐：为每个剪映条目找最近的 Whisper 条目
    for entry in cn_srt:
        best = min(whisper_srt, key=lambda w: abs(w.start - entry.start), default=None)
        if best is None or abs(best.start - entry.start) > 2.0:
            report["red"].append({
                "index": entry.index, "start": entry.start,
                "cn_text": entry.text, "whisper_text": None,
                "reason": "Whisper 无匹配条目"
            })
            continue

        sim = SequenceMatcher(None, entry.text, best.text).ratio()
        detail = {
            "index": entry.index, "start": entry.start,
            "cn_text": entry.text, "whisper_text": best.text,
            "similarity": round(sim, 3),
        }

        if sim >= 0.9:
            report["green"].append(detail)
        elif sim >= 0.7:
            report["yellow"].append(detail)
        else:
            detail["reason"] = f"相似度仅 {sim:.2f}"
            report["red"].append(detail)

    # 检测 Whisper 多出的条目（剪映漏识别）
    cn_starts = {e.start for e in cn_srt}
    for w in whisper_srt:
        if all(abs(w.start - cs) > 2.0 for cs in cn_starts):
            report["red"].append({
                "index": -1, "start": w.start,
                "cn_text": None, "whisper_text": w.text,
                "reason": "剪映缺失此条目"
            })

    return report
```

### 2.6 规则层补充校验

```python
def rule_based_check(entries: list[SrtEntry], glossary: dict[str, str]) -> list[dict]:
    """
    不需要人工肉眼扫的规则自动拦：
    - 人名/地名白名单比对
    - 单条字幕超长 (> 25 字)
    - 字幕持续太短 (< 0.5s) 或太长 (> 8s)
    - 时间轴重叠
    """
    alerts = []
    for e in entries:
        dur = e.end - e.start
        if dur < 0.5:
            alerts.append({"index": e.index, "type": "duration_too_short", "dur": dur, "text": e.text})
        if dur > 8.0:
            alerts.append({"index": e.index, "type": "duration_too_long", "dur": dur, "text": e.text})
        if len(e.text) > 25:
            alerts.append({"index": e.index, "type": "text_too_long", "len": len(e.text), "text": e.text})

    # 术语表检查
    for e in entries:
        for key, expected in glossary.items():
            if key in e.text and expected not in e.text:
                alerts.append({
                    "index": e.index, "type": "glossary_mismatch",
                    "found": key, "expected": expected, "text": e.text,
                })

    # 时间轴重叠检查
    for i in range(len(entries) - 1):
        if entries[i].end > entries[i+1].start:
            alerts.append({
                "type": "overlap",
                "idx_a": entries[i].index, "idx_b": entries[i+1].index,
                "overlap": round(entries[i].end - entries[i+1].start, 2),
            })

    return alerts
```

```python
import httpx

async def translate_srt(entries: list[SrtEntry], target_lang: str,
                        api_key: str, model: str, glossary: dict) -> list[SrtEntry]:
    """
    批量翻译 SRT 字幕。
    核心约束通过 prompt 注入：严禁修改序号、时间轴、条目数量、排列顺序。

    分批策略：每批 40 条，带序号上下文，避免 API 上下文窗口溢出。
    """
    BATCH = 40
    glossary_prompt = "\n".join(f"- {k} → {v}" for k, v in glossary.items())

    system_prompt = f"""你是专业中文→{target_lang}字幕翻译。

术语表（必须遵从）：
{glossary_prompt}

严格规则：
1. 只替换字幕文字，禁止修改序号、时间轴、条目数量、排列顺序
2. 输出标准 SRT 格式，每块用空行分隔
3. 禁止合并、拆分、删除或新增任何字幕条目
4. 使用目标语言自然口语，符合当地表达习惯
5. 保持人物称谓、敬语、语气的准确性"""

    result = []
    for batch_start in range(0, len(entries), BATCH):
        batch = entries[batch_start:batch_start + BATCH]
        batch_text = _entries_to_srt(batch)
        batch_input = f"翻译以下 SRT 字幕为{target_lang}，保持格式不变：\n\n{batch_text}"

        async with httpx.AsyncClient(timeout=120) as client:
            resp = await client.post(
                f"{BASE_URL}/chat/completions",
                headers={"Authorization": f"Bearer {api_key}"},
                json={
                    "model": model,
                    "messages": [
                        {"role": "system", "content": system_prompt},
                        {"role": "user", "content": batch_input},
                    ],
                    "temperature": 0.3,
                },
            )
        data = resp.json()
        translated_text = data["choices"][0]["message"]["content"]
        translated_entries = parse_srt_from_text(translated_text)

        # 校验：条目数必须一致
        if len(translated_entries) != len(batch):
            raise ValueError(f"翻译结果条目数不一致: 期望 {len(batch)}, 实际 {len(translated_entries)}")

        result.extend(translated_entries)

    return result
```

```python
from docx import Document

def generate_review_docx(cn_entries: list[SrtEntry], vi_entries: list[SrtEntry],
                          output_path: Path):
    """
    按序号对齐两份 SRT，生成 Word 表格。
    列：序号 | 时间轴 | 中文字幕 | 越南语初译 | 审核意见 | 修改建议
    """
    doc = Document()
    doc.styles["Normal"].font.size = Pt(10)
    doc.styles["Normal"].font.name = "Arial"

    table = doc.add_table(rows=1, cols=6)
    table.style = "Table Grid"
    table.alignment = WD_TABLE_ALIGNMENT.CENTER

    # 表头
    headers = ["序号", "时间轴", "中文字幕", "越南语初译", "审核意见", "修改建议"]
    for i, h in enumerate(headers):
        cell = table.rows[0].cells[i]
        cell.text = h
        for p in cell.paragraphs:
            p.alignment = 1   # 居中

    # 逐行填充
    cn_map = {e.index: e for e in cn_entries}
    vi_map = {e.index: e for e in vi_entries}

    for idx in sorted(set(cn_map.keys()) | set(vi_map.keys())):
        cn = cn_map.get(idx)
        vi = vi_map.get(idx)
        row = table.add_row()
        row.cells[0].text = str(idx)
        if cn:
            row.cells[1].text = f"{_seconds_to_time(cn.start)} --> {_seconds_to_time(cn.end)}"
            row.cells[2].text = cn.text
        if vi:
            row.cells[3].text = vi.text
        # 审核意见和修改建议留空

    # 列宽
    widths = [Cm(1.2), Cm(3.5), Cm(4), Cm(4), Cm(3), Cm(3)]
    for row in table.rows:
        for i, w in enumerate(widths):
            row.cells[i].width = w

    doc.save(str(output_path))
```

```python
{srt_text}

    async with httpx.AsyncClient(timeout=180) as client:

                    {"role": "user", "content": prompt},
                ],
                "temperature": 0.2,
            },
        )
    data = resp.json()

def _parse_review_docx(path: Path) -> list[dict]:
    """从审核 DOCX 中提取有修改意见的行"""
    doc = Document(str(path))
    items = []
    for table in doc.tables:
        for row in table.rows[1:]:   # 跳过表头
            cells = row.cells
            opinion = cells[4].text.strip() if len(cells) > 4 else ""
            suggestion = cells[5].text.strip() if len(cells) > 5 else ""
            if opinion or suggestion:
                items.append({
                    "index": int(cells[0].text.strip()) if cells[0].text.strip().isdigit() else 0,
                    "cn_text": cells[2].text.strip() if len(cells) > 2 else "",
                    "vi_original": cells[3].text.strip() if len(cells) > 3 else "",
                    "opinion": opinion,
                    "suggestion": suggestion,
                })
    return items
```

## 四、工具函数

```python
def _entries_to_srt(entries: list[SrtEntry]) -> str:
    blocks = []
    for e in entries:
        blocks.append(
            f"{e.index}\n"
            f"{_seconds_to_time(e.start)} --> {_seconds_to_time(e.end)}\n"
            f"{e.text}"
        )
    return "\n\n".join(blocks)


def _seconds_to_time(sec: float) -> str:
    h = int(sec // 3600)
    m = int((sec % 3600) // 60)
    s = int(sec % 60)
    ms = int((sec - int(sec)) * 1000)
    return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}"


def parse_srt_from_text(text: str) -> list[SrtEntry]:
    """从纯文本解析 SRT，兼容 API 返回格式"""
    entries = []
    blocks = text.strip().split("\n\n")
    for block in blocks:
        lines = block.strip().split("\n")
        if len(lines) < 3:
            continue
        try:
            idx = int(lines[0].strip())
        except ValueError:
            continue
        times = re.findall(SRT_TIME, lines[1])
        if len(times) != 2:
            continue
        start = _time_to_seconds(times[0])
        end = _time_to_seconds(times[1])
        txt = "\n".join(lines[2:])
        entries.append(SrtEntry(index=idx, start=start, end=end, text=txt))
    return entries
```
